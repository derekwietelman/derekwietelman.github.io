{{/* Markdown Image Renderer for Hugo Blox Kit. */}} {{/* Load image from page dir falling back to media library at `assets/media/` and then to remote URI. */}} {{/* Site override of the theme's hook (used by the Bio and Contact photos): the largest version is up to 2000px instead of 760px, so phones and tablets get sharp photos, and `sizes` matches this site's layout (full width less the side padding on phones, the ~600px text column on tablets, about 440px wide on desktop) so each screen downloads only the version it needs. A page whose photo is laid out differently (the CV page) sets its own `image_sizes` in its front matter. */}} {{ $destination := .Destination }} {{ $is_remote := strings.HasPrefix $destination "http" }} {{ $caption := .Title | default "" }} {{ $zoom := "true" }} {{ $id := anchorize ($caption | plainify) }} {{ $alt := .Text | default ($caption | plainify) }} {{ $img_class := "" }} {{ $fig_class := "" }} {{ $max_width := "" }} {{ $width := "" }} {{ $height := "" }} {{ $numbered := false }} {{/* Workaround Hugo v0.81 error on Windows when `resources.Get (path.Join "media" )` */}} {{- $img := "" -}} {{- if not $is_remote -}} {{- $img = (.Page.Resources.ByType "image").GetMatch $destination -}} {{- if not $img -}} {{- $img = resources.Get (path.Join "media" $destination) -}} {{- end -}} {{- end -}}
